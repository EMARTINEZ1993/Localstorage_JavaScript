4. Construcción paso a paso: CRUD de estudiantes

Ahora vamos a integrar los conceptos anteriores en un proyecto que permita registrar, consultar, actualizar y eliminar estudiantes.

4.1. Estructura del proyecto

Crea una carpeta para el proyecto y dentro prepara estos archivos. En los siguientes pasos completarás cada uno en orden.

crud-localstorage/

├── index.html

├── css/

4.2. Crear el formulario HTML

Abre index.html y, dentro del elemento body, agrega este formulario. Los campos y mensajes de error tienen identificadores que JavaScript utilizará en los pasos posteriores.

<form id="formulario">

    <label for="nombre">Nombre:</label>
    <input type="text" id="nombre">
    <small id="errorNombre"></small>

    <label for="correo">Correo:</label>
    <input type="text" id="correo">
    <small id="errorCorreo"></small>

    <label for="edad">Edad:</label>
    <input type="number" id="edad">
    <small id="errorEdad"></small>

    <button type="submit" id="btnGuardar">
        Registrar estudiante
    </button>

</form>

4.3. Crear la tabla HTML

Debajo del formulario, agrega una tabla con las columnas de cada dato y las acciones. El elemento tbody vacío será el destino de las filas que JavaScript construirá.

<table>

    <thead>
        <tr>
            <th>Nombre</th>
            <th>Correo</th>
            <th>Edad</th>
            <th>Acciones</th>
        </tr>
    </thead>

    <tbody id="tablaEstudiantes"></tbody>

</table>

4.4. Crear las variables y el estado del proyecto

En js/app.js, conecta los elementos que creaste en el HTML y prepara el arreglo que guardará los estudiantes. El índice -1 indica que el formulario agregará un registro nuevo; un índice distinto se usará al editar.

// 1. VARIABLES Y ESTADO
let formulario = document.getElementById("formulario");

let nombre = document.getElementById("nombre");
let correo = document.getElementById("correo");
let edad = document.getElementById("edad");

let errorNombre = document.getElementById("errorNombre");
let errorCorreo = document.getElementById("errorCorreo");
let errorEdad = document.getElementById("errorEdad");

let tabla = document.getElementById("tablaEstudiantes");
let btnGuardar = document.getElementById("btnGuardar");

let listaEstudiantes = [];

// Variable para controlar la edición
let indiceEditar = -1;

4.5. Validar el nombre

Agrega esta función después de las variables. Comprueba que el nombre no esté vacío y tenga al menos tres caracteres; el evento input la ejecuta mientras se escribe.

// 2. VALIDAR NOMBRE

function validarNombre() {

    let nombreEstudiante = nombre.value.trim();

    if (nombreEstudiante === "") {
        errorNombre.textContent = "Escribe el nombre del estudiante.";
        return false;
    }
    else if (nombreEstudiante.length < 3) {
        errorNombre.textContent = "El nombre debe tener mínimo 3 caracteres.";
        return false;
    }
    else{
        errorNombre.textContent = "";
        return true;
    }

}

nombre.addEventListener("input", validarNombre);

4.6. Validar el correo

A continuación, agrega la validación del correo. La expresión regular verifica un formato básico y el evento input actualiza el mensaje mientras se escribe.

// 3. VALIDAR CORREO

function validarCorreo() {

    let correoEstudiante = correo.value.trim();

    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(correoEstudiante)) {
        errorCorreo.textContent = "Escribe un correo válido.";
        return false;
    }
    else{
        errorCorreo.textContent = "";
        return true;
    }

}

correo.addEventListener("input", validarCorreo);

4.7. Validar la edad

Agrega la validación de edad. Solo se acepta un número entero mayor que cero; más adelante esta misma función se ejecutará otra vez al enviar el formulario.

// 4. VALIDAR EDAD
function validarEdad() {

    let edadEstudiante = Number(edad.value);

    if (edad.value.trim() === "" || !Number.isInteger(edadEstudiante) || edadEstudiante <= 0) {
        errorEdad.textContent = "Escribe una edad entera mayor que cero.";
        return false;
    }
    else{
        errorEdad.textContent = "";
        return true;
    }

}

edad.addEventListener("input", validarEdad);

4.8. Conectar el envío del formulario

Conecta el formulario al evento submit. Al enviarlo, el navegador llamará a registrarEstudiante, cuya función construirás en el siguiente paso.

// 5. EVENTO DEL FORMULARIO

formulario.addEventListener("submit", registrarEstudiante);

4.9. Registrar o actualizar un estudiante

Agrega esta función después de conectar el evento. Primero evita que el formulario recargue la página, valida los campos y crea el objeto. Si no hay una edición activa lo agrega; de lo contrario, reemplaza el registro seleccionado. Finalmente persiste el arreglo y actualiza la tabla.

// 6. REGISTRAR Y ACTUALIZAR ESTUDIANTE

function registrarEstudiante(event) {

    event.preventDefault();

    // Validar los campos antes de registrar
    let nombreValido = validarNombre();
    let correoValido = validarCorreo();
    let edadValida = validarEdad();

    if (!nombreValido || !correoValido || !edadValida) {
        return;
    }

    // Obtener los valores de los campos
    let nombreEstudiante = nombre.value.trim();
    let correoEstudiante = correo.value.trim().toLowerCase();
    let edadEstudiante = Number(edad.value);

    // Crear el objeto del estudiante
    let estudiante = {
        nombre: nombreEstudiante,
        correo: correoEstudiante,
        edad: edadEstudiante
    };

    // Verificar si se está editando un estudiante
    if (indiceEditar === -1) {

        // Agregar un nuevo estudiante al arreglo
        listaEstudiantes.push(estudiante);

    } else {

        // Actualizar el estudiante seleccionado
        listaEstudiantes[indiceEditar] = estudiante;

        indiceEditar = -1;

    }

    // Guardar los datos y actualizar la tabla
    localStorage.setItem("estudiantes", JSON.stringify(listaEstudiantes));

    mostrarEstudiantes();

    // Limpiar el formulario
    formulario.reset();

    // Limpiar los mensajes de error
    errorNombre.textContent = "";
    errorCorreo.textContent = "";
    errorEdad.textContent = "";

    // Restaurar el texto del botón
    btnGuardar.textContent = "Registrar estudiante";

}

4.10. Mostrar los estudiantes en la tabla

Agrega la función que dibuja el arreglo en el tbody. En cada actualización, limpia las filas anteriores y crea una fila por estudiante, con botones vinculados a las acciones correspondientes.

// 7. MOSTRAR ESTUDIANTES EN LA TABLA
function mostrarEstudiantes() {
    tabla.innerHTML = "";

    for (let i = 0; i < listaEstudiantes.length; i++) {

        // Agregar una fila por cada estudiante
        tabla.innerHTML += `
            <tr>
                <td>${listaEstudiantes[i].nombre}</td>
                <td>${listaEstudiantes[i].correo}</td>
                <td>${listaEstudiantes[i].edad}</td>
                <td>
                    <button onclick="editarEstudiante(${i})">
                        Editar
                    </button>
                </td>
                <td>
                    <button onclick="eliminarEstudiante(${i})">
                        Eliminar
                    </button>
                </td>
            </tr>
        `;

    }

}

4.11. Editar un estudiante

Agrega esta función para cargar en el formulario los datos de la fila seleccionada. Guardar su índice permite que el envío del formulario actualice ese mismo objeto en lugar de agregar otro.

// 8. EDITAR ESTUDIANTE

function editarEstudiante(indice) {

    // Cargar los datos del estudiante en los inputs
    nombre.value = listaEstudiantes[indice].nombre;
    correo.value = listaEstudiantes[indice].correo;
    edad.value = listaEstudiantes[indice].edad;

    // Guardar la posición del estudiante seleccionado
    indiceEditar = indice;

    // Cambiar el texto del botón
    btnGuardar.textContent = "Actualizar estudiante";

}

4.12. Eliminar un estudiante

Agrega la función de eliminación. Quita el elemento con splice(), persiste el arreglo y vuelve a dibujar la tabla. También limpia la edición si se elimina ese registro o ajusta el índice si se elimina una fila anterior.

// 9. ELIMINAR ESTUDIANTE


function eliminarEstudiante(indice) {

    // Eliminar el estudiante del arreglo
    listaEstudiantes.splice(indice, 1);

    // Guardar los cambios y actualizar la tabla
    localStorage.setItem("estudiantes", JSON.stringify(listaEstudiantes));
    mostrarEstudiantes();

    // Si se elimina el estudiante que se estaba editando
    if (indiceEditar === indice) {

        indiceEditar = -1;

        formulario.reset();

        btnGuardar.textContent = "Registrar estudiante";

        errorNombre.textContent = "";
        errorCorreo.textContent = "";
        errorEdad.textContent = "";

    }

    // Si se elimina un registro anterior al que se edita
    else if (indiceEditar > indice) {

        indiceEditar--;

    }
}

4.13. Cargar los datos al iniciar

Por último, agrega este bloque al final de js/app.js. Recupera los datos al abrir o recargar la página; si la clave existe, convierte el JSON en arreglo y luego muestra los registros.

// 10. CARGAR DATOS AL INICIAR
// ==========================================

// Consultar los estudiantes guardados
let datosGuardados = localStorage.getItem("estudiantes");

// Verificar si existen datos almacenados
if (datosGuardados !== null) {

    listaEstudiantes = JSON.parse(datosGuardados);

}

// Mostrar los estudiantes en la tabla
mostrarEstudiantes();
